\usepackage{refcount}
\usepackage{hyperref}
\usepackage{bookmark}

% #1 = TOC level, e.g. subsection
% #2 = entry text
% #3 = label identifying the actual destination
\newcommand{\addtocentryat}[3]{%
  % Entry in the printed table of contents:
  \addtocontents{toc}{%
    \protect\contentsline{#1}{#2}%
      {\getpagerefnumber{#3}}%
      {\getrefbykeydefault{#3}{anchor}{}}%
  }%
  % Entry in the PDF document outline:
  \bookmark[
    level=#1,
    dest={\getrefbykeydefault{#3}{anchor}{Doc-Start}}%
  ]{#2}%
}

% An invisible destination, without adding a TOC entry here.
\newcommand{\invisiblemarker}[1]{%
  \phantomsection
  \label{#1}%
}
